+fn hmac(key: &[u8], data: &[u8]) -> [u8; 16] {

+        let words: Vec<u32> = block
+            .as_chunks::<4>()
+            .0
+            .iter()
+            .map(|w| u32::from_le_bytes(*w))
+            .collect();
+        let mut v = state;
+        for round in 0..3 {
+            for step in 0..16 {
+                let a = (16 - step) % 4;
+                let [b, c, d] = [v[(a + 1) % 4], v[(a + 2) % 4], v[(a + 3) % 4]];
+                let (f, k, add, rotation) = match round {
+                    0 => ((b & c) | (!b & d), step, 0, [3, 7, 11, 19][step % 4]),
+                    1 => (
+                        (b & c) | (b & d) | (c & d),
+                        (step % 4) * 4 + step / 4,
+                        0x5a827999,
+                        [3, 5, 9, 13][step % 4],
+                    ),
+                    _ => (
+                        b ^ c ^ d,
+                        [0, 8, 4, 12, 2, 10, 6, 14, 1, 9, 5, 13, 3, 11, 7, 15][step],
+                        0x6ed9eba1,
+                        [3, 9, 11, 15][step % 4],
+                    ),
+                };
+                v[a] = v[a]
+                    .wrapping_add(f)
+                    .wrapping_add(words[k])
+                    .wrapping_add(add)
+                    .rotate_left(rotation);
+            }
+        }
+        for i in 0..4 {
+            state[i] = state[i].wrapping_add(v[i]);
+        }
+    }
+    let mut out = [0; 16];
+    for i in 0..4 {
+        out[i * 4..i * 4 + 4].copy_from_slice(&state[i].to_le_bytes());
+    }
+    out

+pub(crate) fn decode(value: &str) -> Option<Vec<u8>> {
+    let mut out = Vec::new();
+    let mut bits = 0u32;
+    let mut count = 0;
+    for c in value.trim_end_matches('=').bytes() {
+        let n = match c {
+            b'A'..=b'Z' => c - b'A',
+            b'a'..=b'z' => c - b'a' + 26,
+            b'0'..=b'9' => c - b'0' + 52,
+            b'+' => 62,
+            b'/' => 63,
+            _ => return None,
+        };
+        bits = (bits << 6) | u32::from(n);
+        count += 6;
+        if count >= 8 {
+            count -= 8;
+            out.push((bits >> count) as u8);
+        }
+    }
+    Some(out)
