+/// Read the embedded configuration without changing either the file or the active draft.
+pub async fn configuration(app: &App, path: String) -> Result<CrawlConfigDto, CmdError> {
+    // A separate short-lived slot keeps the active session and its cached connections untouched.
+    struct Registration(crate::pool::ReadPool, crate::pool::CrawlKey);
+    impl Drop for Registration {
+        fn drop(&mut self) {
+            self.0.unregister(&self.1);
+        }
+    }
+    let key = crate::pool::CrawlKey(format!(
+        "crawl-file-config-{}",
+        app.config().ids.new_crawl_id(app.config().clock.now())
+    ));
+    let registration = Registration(app.read_pool().clone(), key.clone());
+    registration.0.register(key.clone(), PathBuf::from(path));
+    let config = registration.0.run(key, |file, _| file.config()).await?;
+    super::config::dto(&config)
+}
+
+/// Best-effort recovery into a new file. SQLite recovery itself cannot be interrupted safely.
+pub fn repair(app: &App, path: String) -> OpHandle {
+    let app = app.clone();
+    let ops = app.ops().clone();
+    ops.spawn("crawls-repair", false, move |cancel, progress| {
+        Box::pin(async move {
+            if cancel.is_cancelled() {
+                return Err(CmdError::new(ErrorCode::Cancelled, "err-cancelled"));
+            }
+            progress.report(0, None);
+            let result = app
+                .write_queue()
+                .submit_admin(move || {
+                    os_store::files::repair_copy(&PathBuf::from(path)).map_err(|e| store_error(&e))
+                })
+                .await?;
+            progress.report(1, Some(1));
+            to_json(&serde_json::json!({
+                "path": result.repaired.to_string_lossy(),
+                "integrityOk": result.integrity_ok,
+                "usable": result.usable,
+                "lostRows": result.lost_rows(),
+            }))
+        })
+    })

## Offene Integration und Grenzen

- **C01/U27, OUT-228:** `crates/os-app/src/commands/compare.rs` gehört M2-C01
  und ist weiterhin ein Unsupported-Stub. Dieses Paket ersetzt keine fremde
  Vergleichslogik. Die Aktion bleibt gemäß Fables Empfehlung mit sichtbarer
  Begründung deaktiviert, bis der Vergleichskern und dessen UI-Einstieg wirklich
  verfügbar sind. Bei Integration müssen die Berechtigung für genau zwei
  nicht laufende Crawls, der echte `compare.open`-Aufruf und der folgende
  Vergleichsbildschirm gemeinsam aktiviert und über die Bridge getestet werden.
- Vergleichsfilter vorhanden; gespeicherte benannte Vergleichsreferenzen mit
  Löschwarnung und Projekt-ZIP sind P2 gemäß K2-32/OUT-228.
- Native Dateiadapter benötigen langfristig einen regulären Katalog-/IPC-Eintrag
  durch den Integrator; der eingefrorene Service-Katalog wurde nicht verändert.
- Aufbewahrung, Absturzsicherheit, ID-Vergabe, Migration und portable Dateiinhalte
  liefert O02 bzw. die Crawl-/Store-Schicht, nicht eine zweite Frontend-Implementierung.
- Native OS-Dateidialoge, echte WKWebView/WebView2-Invokes, Windows und
  Referenzhardware-Großläufe wurden hier nicht ausgeführt. Browser verwenden
  deterministische Pfade und eine echte WebSocket-Bridge zum Rust-Kern.
- CI-Lauf-ID: keiner; ausschließlich lokale Prüfung, kein Push/PR beauftragt.
